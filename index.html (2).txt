<script>
(function(){
"use strict";
const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

/* ---------- preloader ---------- */
window.addEventListener("load", () => {
  setTimeout(() => document.getElementById("loader").classList.add("done"), reduced ? 50 : 1400);
});
setTimeout(() => document.getElementById("loader").classList.add("done"), 3200); // failsafe

/* ---------- nav scroll state ---------- */
const nav = document.getElementById("nav");
const onScroll = () => nav.classList.toggle("scrolled", window.scrollY > 40);
window.addEventListener("scroll", onScroll, {passive:true}); onScroll();

/* ---------- mobile menu ---------- */
const burger = document.getElementById("burger"), mm = document.getElementById("mobileMenu");
burger.addEventListener("click", () => {
  const open = mm.classList.toggle("open");
  burger.classList.toggle("open", open);
  burger.setAttribute("aria-expanded", open);
  document.body.style.overflow = open ? "hidden" : "";
});
mm.querySelectorAll("a").forEach(a => a.addEventListener("click", () => {
  mm.classList.remove("open"); burger.classList.remove("open");
  burger.setAttribute("aria-expanded","false"); document.body.style.overflow = "";
}));

/* ---------- reveal on scroll ---------- */
const io = new IntersectionObserver(entries => {
  entries.forEach(e => { if (e.isIntersecting) { e.target.classList.add("in"); io.unobserve(e.target); } });
}, {threshold:.12, rootMargin:"0px 0px -40px 0px"});
document.querySelectorAll(".reveal").forEach(el => io.observe(el));

/* ---------- cursor spotlight ---------- */
const spot = document.getElementById("spotlight");
let sx = innerWidth/2, sy = innerHeight/2, tx = sx, ty = sy;
window.addEventListener("mousemove", e => { tx = e.clientX; ty = e.clientY; }, {passive:true});
(function moveSpot(){
  sx += (tx - sx) * .08; sy += (ty - sy) * .08;
  spot.style.transform = `translate(${sx - 320}px, ${sy - 320}px)`;
  requestAnimationFrame(moveSpot);
})();

/* ---------- magnetic buttons ---------- */
if (!reduced && matchMedia("(pointer:fine)").matches) {
  document.querySelectorAll(".magnetic").forEach(btn => {
    btn.addEventListener("mousemove", e => {
      const r = btn.getBoundingClientRect();
      const x = e.clientX - r.left - r.width/2, y = e.clientY - r.top - r.height/2;
      btn.style.transform = `translate(${x*.18}px, ${y*.28}px)`;
    });
    btn.addEventListener("mouseleave", () => { btn.style.transform = ""; });
  });
}

/* ---------- contact form ---------- */
document.getElementById("contactForm").addEventListener("submit", e => {
  e.preventDefault();
  const f = e.target;
  if (!f.checkValidity()) { f.reportValidity(); return; }
  f.reset();
  document.getElementById("formMsg").classList.add("show");
  setTimeout(() => document.getElementById("formMsg").classList.remove("show"), 5000);
});

/* ---------- THREE.JS hero scene ---------- */
if (window.THREE && !reduced) {
  const canvas = document.getElementById("hero-canvas");
  const renderer = new THREE.WebGLRenderer({canvas, alpha:true, antialias:true});
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(50, 1, .1, 100);
  camera.position.z = 7;

  const group = new THREE.Group();
  scene.add(group);

  // outer glowing wireframe icosahedron
  const geo = new THREE.IcosahedronGeometry(2.15, 1);
  const wire = new THREE.LineSegments(
    new THREE.WireframeGeometry(geo),
    new THREE.LineBasicMaterial({color:0x8b5cf6, transparent:true, opacity:.32})
  );
  group.add(wire);

  // inner solid icosahedron with fresnel-ish glow
  const inner = new THREE.Mesh(
    new THREE.IcosahedronGeometry(1.35, 1),
    new THREE.MeshBasicMaterial({color:0x22d3ee, wireframe:true, transparent:true, opacity:.16})
  );
  group.add(inner);

  // particle field
  const N = 320, pos = new Float32Array(N * 3);
  for (let i = 0; i < N * 3; i++) pos[i] = (Math.random() - .5) * 16;
  const pGeo = new THREE.BufferGeometry();
  pGeo.setAttribute("position", new THREE.BufferAttribute(pos, 3));
  const particles = new THREE.Points(pGeo, new THREE.PointsMaterial({color:0xa78bfa, size:.035, transparent:true, opacity:.65}));
  scene.add(particles);

  // orbiting rings
  const ringMat = new THREE.LineBasicMaterial({color:0x3b82f6, transparent:true, opacity:.22});
  const ring1 = new THREE.LineLoop(new THREE.RingGeometry(3.1, 3.12, 90), ringMat);
  ring1.rotation.x = Math.PI / 2.4;
  const ring2 = new THREE.LineLoop(new THREE.RingGeometry(3.7, 3.72, 90), new THREE.LineBasicMaterial({color:0x22d3ee, transparent:true, opacity:.14}));
  ring2.rotation.x = Math.PI / 1.7; ring2.rotation.y = .5;
  group.add(ring1, ring2);

  // mouse parallax
  let mx = 0, my = 0, cx = 0, cy = 0;
  window.addEventListener("mousemove", e => {
    mx = (e.clientX / innerWidth - .5) * 2;
    my = (e.clientY / innerHeight - .5) * 2;
  }, {passive:true});

  function size() {
    const hero = document.getElementById("hero");
    const w = hero.clientWidth, h = hero.clientHeight;
    renderer.setSize(w, h, false);
    camera.aspect = w / h; camera.updateProjectionMatrix();
    group.position.x = w > 900 ? 2.6 : 0;
    group.position.y = w > 900 ? 0 : 1.2;
    const s = w > 900 ? 1 : .7;
    group.scale.setScalar(s);
  }
  size(); window.addEventListener("resize", size);

  const clock = new THREE.Clock();
  let heroVisible = true;
  new IntersectionObserver(e => heroVisible = e[0].isIntersecting).observe(document.getElementById("hero"));

  (function animate(){
    requestAnimationFrame(animate);
    if (!heroVisible) return;
    const t = clock.getElapsedTime();
    cx += (mx - cx) * .04; cy += (my - cy) * .04;
    group.rotation.y = t * .12 + cx * .5;
    group.rotation.x = cy * .35 + Math.sin(t * .2) * .08;
    inner.rotation.y = -t * .3;
    ring1.rotation.z = t * .18; ring2.rotation.z = -t * .12;
    particles.rotation.y = t * .02;
    renderer.render(scene, camera);
  })();
}
})();
